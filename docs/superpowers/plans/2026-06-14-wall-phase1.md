- [ ] **Step 2: Rewrite drawBuildingWalls**

Replace the entire `drawBuildingWalls` function:

```js
/** Draw south wall objects along building perimeters. Call AFTER F2/player.
 *  South walls are the most visible — full face with windows/doors.
 *  East/west/north walls are Phase 2-3 (honest absence). */
export function drawBuildingWalls(ctx, camX, camY, tilePx, w, h) {
  const buildings = _cache.buildings;
  if (!buildings || buildings.length === 0) return;
  if (!_wallImgs.plain) return;

  ctx.save();
  ctx.imageSmoothingEnabled = false;

  // Each wall segment = exactly 5 tiles (160px source / 32px per tile)
  const SEG_TILES = 5;
  const segSize = Math.round(tilePx * SEG_TILES); // pixel size on screen

  for (const b of buildings) {
    const fp = b.footprint;
    const hasDoor = fp.doors && fp.doors.length > 0;
    const hasWindow = fp.interior?.walls?.some(w2 => w2.kind === 'window');

    for (const sec of fp.sections) {
      const x0 = b.x + sec.x0;
      const y0 = b.y + sec.y0;

      // Only draw south wall: the south edge of this section
      // Skip if the tile below is also a building tile (interior wall, not exterior)
      const southY = y0 + sec.h;

      const numSegs = Math.max(1, Math.ceil(sec.w / SEG_TILES));
      for (let si = 0; si < numSegs; si++) {
        const tileX = x0 + si * SEG_TILES;

        // Screen position: wall sits ON the perimeter border.
        // Bottom edge of wall sprite = south edge of building floor.
        // Wall extends UPWARD from the border.
        const sx = Math.floor(tileX * tilePx - camX);
        const sy = Math.floor(southY * tilePx - camY) - segSize;

        // Clip last segment to section width
        const remainTiles = x0 + sec.w - tileX;
        const drawW = Math.min(segSize, Math.round(remainTiles * tilePx));
        if (drawW <= 0) continue;

        // Screen cull
        if (sx + drawW < 0 || sx > w || sy + segSize < 0 || sy > h) continue;

        // Pick variant: door at center segment, windows at odd segments
        let variant = 'plain';
        const mid = Math.floor(numSegs / 2);
        if (hasDoor && si === mid) variant = 'door';
        else if (hasWindow && si % 2 === 1 && numSegs > 1) variant = 'window';

        const img = _wallImgs[variant];
        if (!img) continue;

        // Source crop: if this is a partial segment (last one), crop from left
        const srcW = Math.round(160 * drawW / segSize);
        ctx.drawImage(img, 0, 0, srcW, 160, sx, sy, drawW, segSize);
      }
    }
  }

  ctx.restore();
}
```

Key positioning logic:
- `sy = southY * tilePx - camY - segSize` — wall's BOTTOM edge aligns with south floor edge, wall extends UPWARD
- `drawW` clips the last segment to fit the section width exactly
- `srcW` crops the source image proportionally for partial segments (no stretching)
