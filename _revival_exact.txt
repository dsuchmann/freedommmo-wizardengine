function paintSwampOverlayRevival(ctx, tile, sx, sy, size) {
  const fertility = tile.climate?.fertility ?? 0.55;
  const vegetation = tile.climate?.vegetation ?? 0.55;
  const moisture = tile.climate?.moisture ?? 0.65;
  const broad = smoothNoise(tile.wx, tile.wy, 16, 16031);
  const mossAlpha = clamp((fertility * 0.36 + vegetation * 0.44 + moisture * 0.20 + broad * 0.25 - 0.58) * 0.42, 0.04, 0.24);
  ctx.save();
  ctx.fillStyle = `rgba(52, 96, 48, ${mossAlpha.toFixed(3)})`;
  ctx.fillRect(sx, sy, size, size);
  const wet = smoothNoise(tile.wx + 11, tile.wy - 7, 9, 16032);
  if (wet > 0.52) {
    ctx.fillStyle = `rgba(25, 42, 34, ${((wet - 0.52) * 0.18).toFixed(3)})`;
    ctx.fillRect(sx, sy, size, size);
  }
  const speck = rand2(tile.wx, tile.wy, 16033);
  if (speck > 0.62) {
    const count = speck > 0.88 ? 2 : 1;
    ctx.fillStyle = 'rgba(79, 121, 55, 0.34)';
    for (let i = 0; i < count; i++) {
      const px = sx + Math.floor(rand2(tile.wx, tile.wy, 16034 + i) * (size - 3));
      const py = sy + Math.floor(rand2(tile.wx, tile.wy, 16044 + i) * (size - 3));
      ctx.fillRect(px, py, Math.max(2, Math.floor(size * 0.10)), 1);
    }
  }
  ctx.restore();
}
