123| function tint(color, tintColor, ambient) {
124|   const rgb = parseRgb(color);
125|   return `rgb(${clamp(rgb.r * tintColor.r * ambient)},${clamp(rgb.g * tintColor.g * ambient)},${clamp(rgb.b * tintColor.b * ambient)})`;
126| }

128| function parseRgb(color) {
129|   if (color.startsWith('rgb')) {
130|     const parts = color.match(/\d+/g).map(Number);
131|     return { r: parts[0], g: parts[1], b: parts[2] };
132|   }
133|   const n = parseInt(color.slice(1), 16);
134|   return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 };
135| }

137| function clamp(value) {
138|   return Math.max(0, Math.min(255, value | 0));
139| }

141| function shade(hex, amount) {
142|   const n = parseInt(hex.slice(1), 16);
143|   let r = (n >> 16) & 255;
144|   let g = (n >> 8) & 255;
145|   let b = n & 255;
146|   r = Math.max(0, Math.min(255, r + amount * 255));
147|   g = Math.max(0, Math.min(255, g + amount * 255));
148|   b = Math.max(0, Math.min(255, b + amount * 255));
149|   return `rgb(${r | 0},${g | 0},${b | 0})`;
150| }
