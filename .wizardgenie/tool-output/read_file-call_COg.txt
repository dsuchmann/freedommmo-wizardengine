125| function preparePixelLabBaseMask(chunk, x, y, tile) {
126|   const family = landscapeRecipe(tile).base.family;
127|   if (family !== 'swamp/wet_mud' && family !== 'swamp/mud_pool') {
128|     preparePixelLabBeachSwampFeather(chunk, x, y, tile);
129|     return;
130|   }
131|   tile.pixelLabSkipBase = false;
132|   tile.pixelLabMissingTransition = '';
133|   tile.pixelLabTransitionBiome = '';
134|   tile.pixelLabTransitionMask = 0;
135|   tile.pixelLabTransitionSides = 0;
136|   tile.pixelLabTransitionSrc = '';
137|   tile.pixelLabTransitionImageMask = 0;
138| 
139|   const neighborTile = (dx, dy) => {
140|     const nx = x + dx;
141|     const ny = y + dy;
142|     if (nx < 0 || ny < 0 || nx >= WORLD.chunkSize || ny >= WORLD.chunkSize) return null;
143|     return chunk.tiles[ny * WORLD.chunkSize + nx] ?? null;
144|   };
145|   const isForeign = (dx, dy) => {
146|     const other = neighborTile(dx, dy);
147|     return Boolean(other?.biome && other.biome !== tile.biome);
148|   };
149| 
150|   const foreignCounts = new Map();
151|   const countForeign = (dx, dy, weight) => {
152|     const other = neighborTile(dx, dy);
153|     if (other?.biome && other.biome !== tile.biome) {
154|       foreignCounts.set(other.biome, (foreignCounts.get(other.biome) ?? 0) + weight);
155|     }
156|   };
157|   countForeign(0, -1, 4);
158|   countForeign(1, 0, 4);
159|   countForeign(0, 1, 4);
160|   countForeign(-1, 0, 4);
161|   countForeign(-1, -1, 1);
162|   countForeign(1, -1, 1);
163|   countForeign(-1, 1, 1);
164|   countForeign(1, 1, 1);
165|   const transitionBiome = [...foreignCounts.entries()].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]))[0]?.[0] ?? '';
166| 
167|   if (transitionBiome) {
168|     const n = isForeign(0, -1);
169|     const e = isForeign(1, 0);
170|     const s = isForeign(0, 1);
171|     const w = isForeign(-1, 0);
172|     const nw = isForeign(-1, -1);
173|     const ne = isForeign(1, -1);
174|     const sw = isForeign(-1, 1);
175|     const se = isForeign(1, 1);
176|     let transitionMask = 0;
177|     // Corner bits are swamp-present bits. Mark a corner as foreign when either
178|     // adjacent side or its diagonal is foreign; this removes the alternating
179|     // diagonal checker pattern from swamp/beach boundaries.
180|     if (!(n || w || nw)) transitionMask |= 8;
181|     if (!(n || e || ne)) transitionMask |= 4;
182|     if (!(s || w || sw)) transitionMask |= 2;
183|     if (!(s || e || se)) transitionMask |= 1;
184|     let sideMask = 0;
185|     if (n) sideMask |= 1;
186|     if (e) sideMask |= 2;
187|     if (s) sideMask |= 4;
188|     if (w) sideMask |= 8;
189|     tile.pixelLabBaseMask = 15;
190|     tile.pixelLabBaseFamily = family;
191|     tile.pixelLabTransitionBiome = transitionBiome;
192|     tile.pixelLabTransitionMask = transitionMask;
193|     tile.pixelLabTransitionSides = sideMask;
194|     tile.pixelLabMissingTransition = `${family}__to__${transitionBiome}`;
195|     return;
196|   }
197| 
198|   const same = (dx, dy) => {
199|     const other = neighborTile(dx, dy);
200|     if (!other) return true;
201|     return landscapeRecipe(other).base.family === family;
202|   };
203|   let mask = 0;
204|   if (same(-1, -1)) mask |= 8;
205|   if (same(1, -1)) mask |= 4;
206|   if (same(-1, 1)) mask |= 2;
207|   if (same(1, 1)) mask |= 1;
208|   tile.pixelLabBaseMask = mask;
209|   tile.pixelLabBaseFamily = family;

212| function preparePixelLabBeachSwampFeather(chunk, x, y, tile) {
