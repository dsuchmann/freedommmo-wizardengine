26|   teleportFromClick(event) {
27|     const rect = this.element.getBoundingClientRect();
28|     const px = Math.floor((event.clientX - rect.left) / rect.width * this.size);
29|     const py = Math.floor((event.clientY - rect.top) / rect.height * this.size);
30|     const { pcx, pcy } = this.playerChunk();
31|     const targetCx = pcx + (px - Math.floor(this.size / 2));
32|     const targetCy = pcy + (py - Math.floor(this.size / 2));

34|     this.player.y = targetCy * WORLD.chunkSize + WORLD.chunkSize / 2;
35|     this.redrawNeeded = true;

39|   draw(force = false) {

42|     const { pcx, pcy } = this.playerChunk();
43|     const centerKey = `${pcx},${pcy}`;

45|       this.drawOverlay(ctx, pcx, pcy);
46|       return;
47|     }
48|     this.lastCenter = centerKey;
49|     this.redrawNeeded = false;
50|     const image = ctx.createImageData(this.size, this.size);
51|     const half = Math.floor(this.size / 2);
52| 
53|     for (let py = 0; py < this.size; py++) {
54|       for (let px = 0; px < this.size; px++) {
55|         const cx = pcx + px - half;
56|         const cy = pcy + py - half;
57|         const wx = Math.floor(cx * WORLD.chunkSize + WORLD.chunkSize / 2);
58|         const wy = Math.floor(cy * WORLD.chunkSize + WORLD.chunkSize / 2);
59|         const sample = classifyBiome(wx, wy);
60|         const rgb = hexToRgb(sample.definition.color);
61|         const hill = Math.max(-35, Math.min(45, (sample.climate.elevation - 0.5) * 95));
62|         const index = (py * this.size + px) * 4;
63|         image.data[index] = clamp(rgb.r + hill);
64|         image.data[index + 1] = clamp(rgb.g + hill);
65|         image.data[index + 2] = clamp(rgb.b + hill);
66|         image.data[index + 3] = 255;
67|       }
68|     }
69| 
70|     ctx.putImageData(image, 0, 0);
71|     this.baseImage = ctx.getImageData(0, 0, this.size, this.size);
72|     this.drawOverlay(ctx, pcx, pcy);
73|   }

137| function hexToRgb(hex) {
138|   const n = parseInt(hex.slice(1), 16);
139|   return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 };
140| }

142| function clamp(value) {
143|   return Math.max(0, Math.min(255, value | 0));
144| }
