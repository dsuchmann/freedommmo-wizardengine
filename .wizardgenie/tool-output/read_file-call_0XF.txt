5| export function paintTerrainTile(ctx, tile, sx, sy, size, sun, focusElevation = tile.climate.elevation, compositor = null, timeSeconds = 0, atlas = null) {
6|   const signature = compositor?.terrainSignature(tile);
7|   const palette = paletteFor(tile.biome);
8|   const micro = rand2(tile.wx, tile.wy, 1200);
9|   const base = palette[Math.min(palette.length - 1, Math.floor(micro * palette.length))];
10|   const elevationShade = (tile.climate.elevation - 0.5) * 0.22;
11|   const depthFade = Math.max(0, focusElevation - tile.climate.elevation - 0.08) * 0.50;
12|   ctx.fillStyle = tint(shade(base, elevationShade + depthFade), sun.tint, sun.ambient);
13|   ctx.fillRect(sx, sy, size, size);
14|   paintBiomeTexture(ctx, tile, sx, sy, size, sun);
15|   paintMicroLayers(ctx, tile, sx, sy, size, sun, timeSeconds, atlas);
16| }

28| function paintBiomeTexture(ctx, tile, sx, sy, size, sun) {
29|   const density = detailDensity(tile.biome);
30|   const count = Math.max(1, Math.floor(size * size * density));
31|   for (let i = 0; i < count; i++) {
32|     const r = rand2(tile.wx, tile.wy, 1300 + i * 17);
33|     const x = sx + Math.floor(rand2(tile.wx, tile.wy, 1400 + i) * size);
34|     const y = sy + Math.floor(rand2(tile.wx, tile.wy, 1500 + i) * size);
35|     if (tile.biome.includes('ocean') || tile.biome === 'shallow_water' || tile.biome === 'river' || tile.biome === 'lake') {
36|       ctx.fillStyle = `rgba(180,230,245,${0.08 + r * 0.10})`;
37|       ctx.fillRect(x, y, Math.max(1, size * 0.35), 1);
38|     } else if (tile.biome === 'forest' || tile.biome === 'dense_forest' || tile.biome === 'taiga' || tile.biome === 'tropical_forest') {
39|       ctx.fillStyle = tint(r > 0.5 ? '#16351f' : '#6f8c3c', sun.tint, sun.ambient * 0.95);
40|       ctx.fillRect(x, y, Math.max(1, size * 0.12), Math.max(1, size * 0.20));
41|     } else if (tile.biome === 'mystic') {
42|     ctx.fillStyle = `rgba(130,90,255,${0.10 + r * 0.18})`;
43|     ctx.fillRect(x, y, Math.max(1, size * 0.16), Math.max(1, size * 0.16));
44|     ctx.fillStyle = `rgba(80,255,220,${0.08 + r * 0.12})`;
45|     ctx.fillRect(x + 1, y, 1, Math.max(1, size * 0.30));
46|   } else if (tile.biome === 'beach' || tile.biome === 'desert') {

48|       ctx.fillRect(x, y, Math.max(1, size * 0.12), 1);

59| function detailDensity(biome) {
60|   if (biome === 'dense_forest' || biome === 'tropical_forest') return 0.014;
61|   if (biome === 'forest' || biome === 'taiga' || biome === 'grassland') return 0.010;
62|   if (biome.includes('ocean') || biome === 'river' || biome === 'lake') return 0.004;
63|   if (biome === 'desert' || biome === 'beach') return 0.005;
64|   return 0.007;
65| }

67| export function shade(hex, amount) {
68|   const n = parseInt(hex.slice(1), 16);
69|   let r = (n >> 16) & 255;
70|   let g = (n >> 8) & 255;
71|   let b = n & 255;
72|   r = Math.max(0, Math.min(255, r + amount * 255));
73|   g = Math.max(0, Math.min(255, g + amount * 255));
74|   b = Math.max(0, Math.min(255, b + amount * 255));
75|   return `rgb(${r | 0},${g | 0},${b | 0})`;
76| }

78| export function tint(color, tintColor, ambient) {
79|   const rgb = parseRgb(color);
80|   return `rgb(${clamp(rgb.r * tintColor.r * ambient)},${clamp(rgb.g * tintColor.g * ambient)},${clamp(rgb.b * tintColor.b * ambient)})`;
81| }

83| function parseRgb(color) {
84|   if (color.startsWith('rgb')) {
85|     const parts = color.match(/\d+/g).map(Number);
86|     return { r: parts[0], g: parts[1], b: parts[2] };
87|   }
88|   const n = parseInt(color.slice(1), 16);
89|   return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 };
90| }

92| function clamp(value) {
93|   return Math.max(0, Math.min(255, value | 0));
