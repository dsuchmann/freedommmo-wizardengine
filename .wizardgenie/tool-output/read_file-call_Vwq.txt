33| export function sampleClimate(wx, wy) {
34|   const continental = fbm(wx, wy, 10);
35|   const detail = fbm(wx, wy, 99);
36|   const elevation = clamp01(continental * 0.88 + detail * 0.30 - 0.06);
37|   const moisture = clamp01(fbm(wx + 9000, wy - 4000, 20) * 1.12);
38|   const latitudeCold = Math.min(0.55, Math.abs(wy) / 9000);
39|   const altitudeCold = Math.max(0, elevation - 0.58) * 0.55;
40|   const heat = clamp01(fbm(wx - 3000, wy + 7000, 30) * 1.12 - latitudeCold - altitudeCold + 0.08);
41|   const drainage = fbm(wx + 17000, wy - 11000, 40);
42|   return { elevation, moisture, heat, drainage };
43| }

45| export function classifyBiome(wx, wy) {
46|   const climate = sampleClimate(wx, wy);
47|   const { elevation, moisture, heat, drainage } = climate;
48|   let id;
49|   if (elevation < 0.26) id = 'deep_ocean';
50|   else if (elevation < 0.36) id = 'ocean';
51|   else if (elevation < 0.40) id = 'shallow_water';
52|   else if (elevation < 0.42) id = moisture > 0.82 ? 'lake' : 'beach';
53|   else if (drainage > 0.73 && moisture > 0.58 && elevation < 0.62) id = 'river';
54|   else if (moisture > 0.86 && elevation < 0.58) id = 'lake';
55|   else if (elevation > 0.74 && heat > 0.36 && moisture < 0.56 && drainage > 0.58) id = 'volcanic';
56|   else if (elevation > 0.78) id = heat < 0.24 ? 'arctic' : 'mountains';
57|   else if (elevation > 0.66) id = heat < 0.30 ? 'tundra' : 'hills';
58|   else if (heat < 0.22) id = moisture > 0.45 ? 'taiga' : 'tundra';
59|   else if (moisture > 0.80 && heat > 0.58) id = 'tropical_forest';
60|   else if (moisture > 0.76 && heat > 0.32) id = 'swamp';
61|   else if (moisture > 0.66) id = heat > 0.48 ? 'dense_forest' : 'forest';
62|   else if (moisture < 0.25 && heat > 0.52) id = 'desert';
63|   else if (moisture < 0.34 && heat > 0.34) id = 'steppe';
64|   else if (moisture < 0.44 && heat > 0.48) id = 'savanna';
65|   else id = 'grassland';
66|   return { id, definition: BIOMES[id], climate };
67| }
