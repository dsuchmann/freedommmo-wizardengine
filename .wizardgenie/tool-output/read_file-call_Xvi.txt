99| function shade(hex, amount) {
100|   const n = parseInt(hex.slice(1), 16);
101|   let r = (n >> 16) & 255;
102|   let g = (n >> 8) & 255;
103|   let b = n & 255;
104|   r = Math.max(0, Math.min(255, r + amount * 255));
105|   g = Math.max(0, Math.min(255, g + amount * 255));
106|   b = Math.max(0, Math.min(255, b + amount * 255));
107|   return `rgb(${r | 0},${g | 0},${b | 0})`;
108| }
