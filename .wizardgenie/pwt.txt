function paintWangTransition(ctx, tile, mask, sx, sy, size, chunk, x, y) {
  if (mask === 15 || mask < 0 || mask > 15) return;
  const half = size / 2;
  ctx.save();
  ctx.globalAlpha = 0.35;
  // Corner definitions: [bit, quadX, quadY, diagDX, diagDY]
  const quads = [
    [8,  sx,        sy,        -1, -1],
    [4,  sx+half,   sy,        +1, -1],
    [2,  sx,        sy+half,   -1, +1],
    [1,  sx+half,   sy+half,   +1, +1],
  ];
  for (let i = 0; i < quads.length; i++) {
    const q = quads[i];
    if (mask & q[0]) continue; // same biome — no transition needed
    // Look up the NEIGHBOR biome's palette color at this diagonal corner
    const nb = safeBiome(chunk, chunk.cx, chunk.cy, x + q[3], y + q[4], tile.biome);
    const nc = paletteColor(nb, 1);
    const nr = parseInt(nc.slice(1,3),16), ng = parseInt(nc.slice(3,5),16), nbv = parseInt(nc.slice(5,7),16);
    const blend = 'rgba('+Math.max(0,nr-30)+','+Math.max(0,ng-30)+','+Math.max(0,nbv-30)+',0.50)';
    // Gradient origin at the outer corner of this quadrant
    const gx = (q[0] === 4 || q[0] === 1) ? sx + size : sx;
    const gy = (q[0] === 2 || q[0] === 1) ? sy + size : sy;
    const grad = ctx.createRadialGradient(gx, gy, 0, gx, gy, size * 0.6);
    grad.addColorStop(0, blend);
    grad.addColorStop(1, 'transparent');
    ctx.fillStyle = grad;
    ctx.fillRect(q[1], q[2], half, half);
  }
