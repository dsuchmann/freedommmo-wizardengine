  const target = PIXEL_BASE_TARGET_RGB[baseFamily];
  if (!target) return false;
  const broad = smoothNoise(tile.wx, tile.wy, 18, 16021) - 0.5;
  const fine = smoothNoise(tile.wx, tile.wy, 7, 16022) - 0.5;
  const fertility = tile.climate?.fertility ?? 0.55;
  const vegetation = tile.climate?.vegetation ?? 0.55;
  const moisture = tile.climate?.moisture ?? 0.65;
  const moss = clamp((fertility * 0.45 + vegetation * 0.40 + moisture * 0.25 - 0.38) * 0.72, 0.08, 0.38);
  const shadeAmount = broad * 16 + fine * 5;
  const mossRgb = [62, 91, 48];
  const wetRgb = [42, 54, 38];
  const wet = clamp((moisture - 0.52) * 0.28, 0, 0.16);
  let r = (target[0] + shadeAmount * 0.78) * (1 - moss) + mossRgb[0] * moss;
  let g = (target[1] + shadeAmount * 0.58) * (1 - moss) + mossRgb[1] * moss;
  let b = (target[2] + shadeAmount * 0.42) * (1 - moss) + mossRgb[2] * moss;
  r = clamp(r * (1 - wet) + wetRgb[0] * wet, 0, 255);
  g = clamp(g * (1 - wet) + wetRgb[1] * wet, 0, 255);
  b = clamp(b * (1 - wet) + wetRgb[2] * wet, 0, 255);
  ctx.fillStyle = `rgb(${Math.round(r)},${Math.round(g)},${Math.round(b)})`;
  ctx.fillRect(sx, sy, size, size);
  return true;
}

function paintSwampTransitionGradient(ctx, tile, sx, sy, size) {
  const mask = Number.isInteger(tile.pixelLabTransitionMask) ? tile.pixelLabTransitionMask : 0;
  const sideMask = Number.isInteger(tile.pixelLabTransitionSides) ? tile.pixelLabTransitionSides : 0;
  const other = PIXEL_TRANSITION_TARGET_RGB[tile.pixelLabTransitionBiome];
  const swampBase = PIXEL_BASE_TARGET_RGB[tile.pixelLabBaseFamily] ?? PIXEL_BASE_TARGET_RGB['swamp/wet_mud'];
  const swamp = [swampBase[0] * 0.78 + 62 * 0.22, swampBase[1] * 0.78 + 91 * 0.22, swampBase[2] * 0.78 + 48 * 0.22];
  if (!mask || !other || !swamp) return false;
  const mix = (a, b, t) => [a[0] * (1 - t) + b[0] * t, a[1] * (1 - t) + b[1] * t, a[2] * (1 - t) + b[2] * t];
  const rgba = (c, alpha = 1) => `rgba(${Math.round(c[0])},${Math.round(c[1])},${Math.round(c[2])},${alpha})`;
  const beachSide = tile.biome === 'beach' || tile.biome === 'desert';
  const edge = mix(swamp, other, beachSide ? 0.42 : 0.46);
  const lip = mix(swamp, other, beachSide ? 0.28 : 0.24);
  const band = size * (beachSide ? 0.18 : 0.40);
  const edgeAlpha = beachSide ? 0.10 : 0.38;
  const lipAlpha = beachSide ? 0.05 : 0.18;

  const fillLinear = (x0, y0, x1, y1, rect, color, alpha) => {
    const gradient = ctx.createLinearGradient(x0, y0, x1, y1);
    gradient.addColorStop(0, rgba(color, alpha));
    gradient.addColorStop(0.58, rgba(color, alpha * 0.42));
    gradient.addColorStop(1, rgba(color, 0));
    ctx.fillStyle = gradient;
    ctx.fillRect(rect.x, rect.y, rect.w, rect.h);
  };
  const fillRadial = (cx, cy, radius, color, alpha) => {
    const gradient = ctx.createRadialGradient(cx, cy, 0, cx, cy, radius);
    gradient.addColorStop(0, rgba(color, alpha));
    gradient.addColorStop(0.52, rgba(color, alpha * 0.38));
    gradient.addColorStop(1, rgba(color, 0));
    ctx.fillStyle = gradient;
    ctx.fillRect(sx, sy, size, size);
  };

  ctx.save();
  // Gradients replace the former polygon stamps. They still respect Wang-side
  // metadata, but no longer leave full tile-width trapezoids along diagonal
  // swamp/sand borders.
  if (sideMask & 1) fillLinear(sx, sy, sx, sy + band, { x: sx, y: sy, w: size, h: band }, edge, edgeAlpha);
  if (sideMask & 2) fillLinear(sx + size, sy, sx + size - band, sy, { x: sx + size - band, y: sy, w: band, h: size }, edge, edgeAlpha);
  if (sideMask & 4) fillLinear(sx, sy + size, sx, sy + size - band, { x: sx, y: sy + size - band, w: size, h: band }, edge, edgeAlpha);
  if (sideMask & 8) fillLinear(sx, sy, sx + band, sy, { x: sx, y: sy, w: band, h: size }, edge, edgeAlpha);

  if (!beachSide) {
    const cornerAlpha = 0.20;
    if (!(mask & 8)) fillRadial(sx, sy, size * 0.46, edge, cornerAlpha);
    if (!(mask & 4)) fillRadial(sx + size, sy, size * 0.46, edge, cornerAlpha);
    if (!(mask & 2)) fillRadial(sx, sy + size, size * 0.46, edge, cornerAlpha);
    if (!(mask & 1)) fillRadial(sx + size, sy + size, size * 0.46, edge, cornerAlpha);
  }

  const lipBand = size * (beachSide ? 0.10 : 0.18);
  if (sideMask & 1) fillLinear(sx, sy, sx, sy + lipBand, { x: sx, y: sy, w: size, h: lipBand }, lip, lipAlpha);
  if (sideMask & 2) fillLinear(sx + size, sy, sx + size - lipBand, sy, { x: sx + size - lipBand, y: sy, w: lipBand, h: size }, lip, lipAlpha);
  if (sideMask & 4) fillLinear(sx, sy + size, sx, sy + size - lipBand, { x: sx, y: sy + size - lipBand, w: size, h: lipBand }, lip, lipAlpha);
  if (sideMask & 8) fillLinear(sx, sy, sx + lipBand, sy, { x: sx, y: sy, w: lipBand, h: size }, lip, lipAlpha);
  ctx.restore();
  return true;
}

function paintSwampOverlayRevival(ctx, tile, sx, sy, size) {
  const fertility = tile.climate?.fertility ?? 0.55;
  const vegetation = tile.climate?.vegetation ?? 0.55;
  const moisture = tile.climate?.moisture ?? 0.65;
  const broad = smoothNoise(tile.wx, tile.wy, 16, 16031);
  const mossAlpha = clamp((fertility * 0.36 + vegetation * 0.44 + moisture * 0.20 + broad * 0.25 - 0.58) * 0.42, 0.04, 0.24);
  ctx.save();
  ctx.fillStyle = `rgba(52, 96, 48, ${mossAlpha.toFixed(3)})`;
  ctx.fillRect(sx, sy, size, size);
  const wet = smoothNoise(tile.wx + 11, tile.wy - 7, 9, 16032);
  if (wet > 0.52) {
    ctx.fillStyle = `rgba(25, 42, 34, ${((wet - 0.52) * 0.18).toFixed(3)})`;
    ctx.fillRect(sx, sy, size, size);
  }
  const speck = rand2(tile.wx, tile.wy, 16033);
  if (speck > 0.62) {
    const count = speck > 0.88 ? 2 : 1;
    ctx.fillStyle = 'rgba(79, 121, 55, 0.34)';
    for (let i = 0; i < count; i++) {
      const px = sx + Math.floor(rand2(tile.wx, tile.wy, 16034 + i) * (size - 3));
      const py = sy + Math.floor(rand2(tile.wx, tile.wy, 16044 + i) * (size - 3));
      ctx.fillRect(px, py, Math.max(2, Math.floor(size * 0.10)), 1);
    }
  }
  ctx.restore();
}

function drawSwampDecal(ctx, family, tile, sx, sy, size, salt, alpha, scale = 1, xBias = 0.5, yBias = 0.5) {
  const variants = PIXEL_SWAMP_DECALS[family];
  if (!variants?.length) return false;
  const index = Math.floor(rand2(tile.wx, tile.wy, salt) * variants.length) % variants.length;
  const image = pixelBaseImage(variants[index]);
  if (!image?.complete || !(image.naturalWidth || image.width)) return false;
  const previousSmoothing = ctx.imageSmoothingEnabled;
  const previousAlpha = ctx.globalAlpha;
  ctx.imageSmoothingEnabled = true;
  ctx.globalAlpha = previousAlpha * alpha;
  const w = size * scale * (0.75 + rand2(tile.wx, tile.wy, salt + 1) * 0.50);
  const h = size * scale * (0.75 + rand2(tile.wx, tile.wy, salt + 2) * 0.50);
  const xJitter = (rand2(tile.wx, tile.wy, salt + 3) - 0.5) * size * 0.72;
  const yJitter = (rand2(tile.wx, tile.wy, salt + 4) - 0.5) * size * 0.72;
  const dx = sx + size * xBias + xJitter - w / 2;
  const dy = sy + size * yBias + yJitter - h / 2;
  const rotate = (Math.floor(rand2(tile.wx, tile.wy, salt + 5) * 4) & 3) * Math.PI / 2;
  const flip = rand2(tile.wx, tile.wy, salt + 6) > 0.5;
  ctx.save();
  ctx.beginPath();
  ctx.rect(sx, sy, size, size);
  ctx.clip();
  ctx.translate(dx + w / 2, dy + h / 2);
  ctx.rotate(rotate);
  ctx.scale(flip ? -1 : 1, 1);
  ctx.drawImage(image, -w / 2, -h / 2, w, h);
  ctx.restore();
  ctx.imageSmoothingEnabled = previousSmoothing;
  ctx.globalAlpha = previousAlpha;
  return true;
}
